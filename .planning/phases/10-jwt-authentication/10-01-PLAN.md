3. Создать `data_ingress/src/web/dependencies/auth.py` — get_current_user dependency:
   ```python
   from typing import Annotated
   import jwt
   from jwt.exceptions import InvalidTokenError
   from fastapi import Depends, HTTPException, status
   from fastapi.security import OAuth2PasswordBearer
   from src.configs.auth import auth_settings

   oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/auth/login")

   async def get_current_user(
       token: Annotated[str, Depends(oauth2_scheme)],
   ) -> int:
       credentials_exception = HTTPException(
           status_code=status.HTTP_401_UNAUTHORIZED,
           detail="Could not validate credentials",
           headers={"WWW-Authenticate": "Bearer"},
       )
       try:
           payload = jwt.decode(token, auth_settings.JWT_SECRET, algorithms=[auth_settings.JWT_ALGORITHM])
           user_id: int | None = payload.get("sub")
           if user_id is None:
               raise credentials_exception
       except InvalidTokenError:
           raise credentials_exception
       return user_id
   ```
