class UnitOfWork(AbstractUnitOfWork):
    def __init__(self): ...  # Creates RabbitMQPublisher, sets _s3=None, _s3_uploaded_keys=[], _pending_messages=[]
    async def __aenter__(self): ...  # Creates session from _session_factory, RecordingRepository
